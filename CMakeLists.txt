cmake_minimum_required(VERSION 3.16)
project(depcover LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(DEPCOVER_SANITIZE "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(DEPCOVER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_library(depcover src/depcover.cpp)
target_include_directories(depcover PUBLIC include)
target_compile_options(depcover PRIVATE -Wall -Wextra -Wpedantic)

add_executable(benchmark benchmarks/benchmark_main.cpp)
target_link_libraries(benchmark PRIVATE depcover)
target_compile_options(benchmark PRIVATE -Wall -Wextra -Wpedantic -O3)
target_compile_definitions(benchmark PRIVATE NDEBUG)
set_target_properties(benchmark PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/benchmarks)

add_executable(depcover_tests tests/test_main.cpp)
target_link_libraries(depcover_tests PRIVATE depcover)
target_compile_options(depcover_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME depcover_tests COMMAND depcover_tests)
